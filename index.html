async function sendToEMS() { const blob = new Blob(audioChunks, { type: "audio/webm" }); const reader = new FileReader(); reader.readAsDataURL(blob); reader.onloadend = async () => { const base64Audio = reader.result.split(",")[1]; try { const response = await fetch(EMS_URL, { method: "POST", // ↓↓↓ THIS IS THE KEY CHANGE ↓↓↓ headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "aiNoteTaker", audio: base64Audio, mimeType: "audio/webm" }), redirect: "follow" // important for GAS redirects }); const text = await response.text(); let result; try { result = JSON.parse(text); } catch (e) { throw new Error("Invalid response from server: " + text.substring(0, 200)); } if (result.success) { document.getElementById("status").textContent = "Done!"; document.getElementById("transcriptBox").style.display = "block"; document.getElementById("transcriptBox").textContent = result.transcript; if (result.docUrl) { document.getElementById("openDocBtn").style.display = "block"; document.getElementById("openDocBtn").href = result.docUrl; } } else { throw new Error(result.message || "Unknown error from server"); } } catch (err) { document.getElementById("status").textContent = "Error occurred"; document.getElementById("transcriptBox").style.display = "block"; document.getElementById("transcriptBox").textContent = "Failed: " + err.message; console.error(err); } }; }
